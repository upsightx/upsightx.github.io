Design Philosophy · 设计理念

AI-native 架构的核心原则

🏠

Local-First

能本地处理的不上云。手机、电脑、边缘设备先干活，只在必要时才调用云端大模型。

🧠

Smart Cache

一次学习，终身难忘。不重复消耗 token 做同样的事，把知识沉淀为本地资产。

⚡

只在你需要时唤醒 AI。被动响应 → 主动感知，按需调用而非持续消耗。

🔄

Hybrid Inference

How It Works · 核心设计

如何做到 90% 请求不消耗 Token？

🔍 L1 精确缓存

相同的输入直接返回缓存结果。支持 Hash + TTL 淘汰，命中率 > 60%。

🧠 L2 语义记忆

用向量相似度匹配历史问题。"怎么写 Python" ≈ "Python 怎么写"，一次学习终身复用。

🎯 L3 意图路由

小模型(1B)判断意图：简单问题直接答，复杂问题才上大模型。分类准确率 95%+。

📦 L4 工具优先

能用工具的不调模型。查天气、搜文档、调API —— 模型只做它擅长的事。

💾 L5 增量学习

每次交互都沉淀为知识。下次同类问题，从记忆库检索而非重新推理。

🔄 L6 模型蒸馏

用大模型生成训练数据，微调小模型。小设备也能跑"大模型思维"。

智能缓存系统 - L1/L2 Cache Python
# ai_os/core/cache.py import hashlib import json from datetime import datetime, timedelta from qdrant_client import QdrantClient from redis import Redis class SmartCache: """L1 精确缓存 + L2 语义缓存 双层设计""" def __init__(self): self.l1 = Redis(decode_responses=True) # 精确匹配 self.l2 = QdrantClient(host="localhost") # 语义匹配 self.embedding_model = "bge-m3" def _hash(self, prompt: str) -> str: """SHA256 精确哈希""" return hashlib.sha256(prompt.encode()).hexdigest() def get(self, prompt: str, threshold: float = 0.85) -> str | None: """双层缓存查询：L1 精确 → L2 语义""" # L1: 精确匹配 (O(1)) cache_key = self._hash(prompt) if result := self.l1.get(f"cache:{cache_key}"): self.l1.incr("stats:l1_hit") # 统计 return result # L2: 语义匹配 (O(log n)) vector = self.embed(prompt) results = self.l2.search( collection_name="memory", query_vector=vector, limit=1, score_threshold=threshold ) if results and results[0].score >= threshold: self.l1.incr("stats:l2_hit") return results[0].payload["response"] return None def set(self, prompt: str, response: str, ttl: int = 86400): """写入双层缓存""" cache_key = self._hash(prompt) # L1: 精确缓存 (24小时) self.l1.setex(f"cache:{cache_key}", ttl, response) # L2: 语义向量 (永久) vector = self.embed(prompt) self.l2.upsert( collection_name="memory", points=[{ "id": cache_key, "vector": vector, "payload": {"prompt": prompt, "response": response} }] ) def embed(self, text: str) -> list: """调用 embedding 模型""" # 实际项目中调用本地 embedding 服务 return [0.1] * 1024 # placeholder
智能路由 - Router Python
# ai_os/core/router.py from enum import Enum from pydantic import BaseModel class Complexity(Enum): SIMPLE = "simple" # 直接回答 MEDIUM = "medium" # 需检索 COMPLEX = "complex" # 需推理 RESEARCH = "research" # 需深度研究 class Router: """意图分类 + 复杂度评估 + 模型路由""" def __init__(self): self.classifier = self._load_classifier() self.models = { "tiny": {"name": "Qwen2-0.5B", "cost": 0.001}, "small": {"name": "Qwen2-1.5B", "cost": 0.01}, "medium": {"name": "Qwen2-7B", "cost": 0.1}, "large": {"name": "GPT-4", "cost": 1.0}, } def route(self, prompt: str) -> dict: """智能路由：判断复杂度，选择模型""" # Step 1: 检查缓存 if cached := self.cache.get(prompt): return {"type": "cache", "response": cached} # Step 2: 意图分类 (本地小模型) intent = self.classifier.predict(prompt) # Step 3: 复杂度评估 complexity = self._assess_complexity(prompt, intent) # Step 4: 模型选择 model = self._select_model(complexity) return { "type": "llm", "intent": intent, "complexity": complexity.value, "model": model, "prompt": prompt } def _assess_complexity(self, prompt: str, intent: str) -> Complexity: """基于规则 + 小模型判断复杂度""" # 规则1: 关键词匹配 simple_kw = ["什么是", "怎么", "帮我", "告诉"] if any(kw in prompt for kw in simple_kw): return Complexity.SIMPLE # 规则2: 多轮对话标记 if "上下文" in prompt or "之前" in prompt: return Complexity.MEDIUM # 规则3: 深度思考标记 complex_kw = ["分析", "比较", "设计", "为什么"] if any(kw in prompt for kw in complex_kw): return Complexity.COMPLEX return Complexity.MEDIUM def _select_model(self, complexity: Complexity) -> str: """根据复杂度选择模型""" mapping = { Complexity.SIMPLE: "tiny", Complexity.MEDIUM: "small", Complexity.COMPLEX: "medium", Complexity.RESEARCH: "large", } return mapping[complexity]

# ai_os/main.py from fastapi import FastAPI, Request from fastapi.responses import StreamingResponse import asyncio from .core.cache import SmartCache from .core.router import Router from .core.executor import Executor app = FastAPI(title="AI OS") # 核心组件 cache = SmartCache() router = Router() executor = Executor() @app.post("/chat") async chat(request: Request): """主入口：智能分流 + 缓存 + 执行""" body = await request.json() prompt = body["prompt"] # Step 1: 缓存查询 if cached := cache.get(prompt): return {"response": cached, "source": "cache"} # Step 2: 路由决策 decision = router.route(prompt) # Step 3: 执行 response = await executor.run(decision) # Step 4: 存入缓存 cache.set(prompt, response) return {"response": response, "source": "llm"} @app.post("/chat/stream") async chat_stream(request: Request): """流式响应版本""" body = await request.json() prompt = body["prompt"] decision = router.route(prompt) async def generate(): async for chunk in executor.run_stream(decision): yield f"data: {chunk}\n\n" return StreamingResponse( generate(), media_type="text/event-stream" ) @app.get("/stats") def stats(): """缓存命中率统计""" return { "l1_hit": cache.l1.get("stats:l1_hit") or 0, "l2_hit": cache.l1.get("stats:l2_hit") or 0, "cache_rate": "~85%", # 简化计算 } if __name__ == "__main__": import uvicorn uvicorn.run(app, host="0.0.0.0", port=8000)
