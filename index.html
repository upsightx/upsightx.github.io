# ai_os/agent.py import asyncio from datetime import datetime from collections import defaultdict class CloudAgent: """ 云端智能体：只做两件事 1. 接收请求 2. 调用 LLM + 工具链 """ def __init__(self, provider: str = "minimax"): self.provider = provider self.memory = [] # 对话历史 self.tools = {} # 注册的工具 self.usage = defaultdict(int) # 用量统计 @property def cost(self) -> float: """当前会话费用""" rates = {"minimax": 0.5, "openai": 15.0} total = self.usage["input"] + self.usage["output"] return (total / 1_000_000) * rates.get(self.provider, 10) def register_tool(self, name: str, func: callable): """注册一个工具：搜索、计算、执行任意操作""" self.tools[name] = { "func": func, "description": func.__doc__ or "" } async def run(self, prompt: str, stream: bool = False) -> dict: """核心流程：理解 → 规划 → 执行 → 响应""" # 1. 构建上下文（包含历史记忆） context = self._build_context(prompt) # 2. LLM 判断意图 + 生成计划 plan = await self._llm_call( context, system="""你是一个AI助手。分析用户请求，""" """决定是否需要调用工具。""" """可用工具: """ + ", ".join(self.tools.keys()) ) # 3. 执行工具（如需要） if tool_calls := plan.get("tool_calls"): results = [] for call in tool_calls: tool_name = call["name"] args = call["arguments"] result = await self.tools[tool_name]["func"](**args) results.append({tool_name: result}) # 4. 用工具结果生成最终回复 final = await "\n\nTool Results:\n" + str(results), system="基于工具结果，给出最终回答" ) else: final = plan["response"] # 5. 存入记忆 + 返回 self.memory.append({ "role": "user", "content": prompt }) self.memory.append({ "role": "assistant", "content": final }) return { "response": final, "cost": "tools_used": [c["name"] for c in tool_calls] if tool_calls else [] } def _build_context(self, prompt: str) -> str: """构建带记忆的上下文（自动裁剪）""" # 保留最近 N 轮对话，节省 token recent = self.memory[-10:] history = "\n".join([f"{m['role']}: {m['content']}" for m in recent]) return f"{history}\nuser: {prompt}" async def _llm_call(self, prompt: str, system: str = "") -> dict: """调用云端 LLM（这里用 MiniMax 示例）""" # 实际项目中调用 MiniMax / OpenAI / Anthropic API return {"response": "LLM 回复内容"} # 使用示例 agent = CloudAgent(provider="minimax") # 注册工具 import httpx @agent.register_tool async def search_web(query: str) -> str: """搜索互联网""" async with httpx.AsyncClient() as client: # 调用搜索 API... return "搜索结果..." # 运行 result = await agent.run("帮我查一下最近的AI新闻") print(result)
