# 100亿Token/年算力预算 = 每月约8.3B from aios import AIOS, Memory, Agent, Tool # 1. 初始化AI OS（按预算配置） aios = AIOS( monthly_budget=8_300_000_000, # 100亿/年 model_routing="cost-optimized", cache_policy="aggressive" ) # 2. 加载个人记忆（向量化存储） memory = Memory( sources=["notion", "email", "slack", "docs"], embed_model="bge-m3", vector_db="qdrant" ) # 3. 定义Agent能力 @Agent(name="researcher", priority="high") def research(topic: str, depth: str = "deep"): """深度研究：搜索、阅读、总结、产出报告""" results = web_search(topic) summaries = [read(url) for url in results] report = summarize(summaries, style="executive") memory.save("research", topic, report) return report # 4. 启动（24/7运行） aios.serve()
